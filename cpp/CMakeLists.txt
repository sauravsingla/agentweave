cmake_minimum_required(VERSION 3.16)
project(agentweave_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(agentweave_core STATIC src/agentweave_core.cpp)
target_include_directories(agentweave_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
option(AGENTWEAVE_BUILD_PYBIND "Build optional Python bindings" OFF)
if(AGENTWEAVE_BUILD_PYBIND)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_agentweave_core bindings.cpp)
  target_link_libraries(_agentweave_core PRIVATE agentweave_core)
endif()
